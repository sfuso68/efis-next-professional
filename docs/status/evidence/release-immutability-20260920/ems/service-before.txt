# /etc/systemd/system/efis-ems.service
[Unit]
Description=EFIS Next Engine Monitoring Node
Documentation=file:///home/pi/efis-next/docs/deployment/RPI4_EMS.md
After=multi-user.target
StartLimitIntervalSec=0

# Unita' di sistema, non utente, e percorsi espliciti: e' la stessa forma che
# efis.service usa su questo nodo per l'EFIS vecchio. Chi mantiene la macchina
# conosce gia' quella forma, e farne una diversa per il solo gusto di essere
# piu' generici costringerebbe a impararne due. Un servizio utente avrebbe
# richiesto in piu' `loginctl enable-linger` per partire senza login, cioe' un
# pezzo di configurazione non versionato.
#
# Nessuna dipendenza da graphical-session.target: il nodo disegna con eglfs,
# cioe' direttamente su KMS/DRM, e non c'e' compositore da attendere.
#
# Nessun Conflicts con efis.service, per scelta esplicita del 21 agosto 2026.
# I due programmi non possono pero' tenere il pannello DSI insieme, perche'
# eglfs prende il DRM master in esclusiva: per riavviare l'EFIS vecchio a mano
# bisogna prima fermare questo nodo. La procedura e' in RPI4_EMS.md.

[Service]
Type=simple
User=pi
WorkingDirectory=/home/pi/efis-next
Environment=HOME=/home/pi
Environment=QT_QPA_PLATFORM=eglfs
ExecStart=/home/pi/efis-next/deploy/scripts/run-ems.sh
# Restart=always, non on-failure, e la ragione e' stata scoperta rompendo il
# nodo il 21 agosto 2026. Un SIGTERM arrivato da fuori — un pkill sbagliato, un
# gestore di memoria, chiunque — e' indistinguibile da un arresto voluto, e con
# on-failure sommato al SuccessExitStatus qui sotto il servizio si spegneva e
# non tornava piu': pannello nero a tempo indeterminato, senza che nulla
# risultasse guasto. Per un nodo il cui unico compito e' mostrare qualcosa,
# quello e' il guasto peggiore, perche' non si annuncia.
#
# E' anche cio' che efis.service fa gia' su questa macchina, per lo stesso
# motivo. Il prezzo e' che un binario davvero rotto viene riprovato all'
# infinito ogni cinque secondi: preferibile a un pannello spento in silenzio, e
# visibile nel journal.
Restart=always
RestartSec=5

# Su eglfs Qt installa un gestore per il terminale virtuale che, ricevuto
# SIGTERM, fa uscire il programma con codice 1. Senza questa riga ogni arresto
# voluto lascerebbe l'unita' in stato failed, cioe' un arresto normale
# somiglierebbe a un guasto. Non e' una particolarita' del nodo EMS: il 21
# agosto 2026 il journal mostra che anche efis.service, l'EFIS vecchio, esce
# con status=1/FAILURE ogni volta che viene fermato, sulla stessa piattaforma.
#
# Perche' la riga non nasconda un guasto vero, il programma non usa il codice 1
# per nessun proprio fallimento: 3 e' la pagina QML non caricata, 2 la cattura
# fallita, e restano entrambi guasti agli occhi di systemd.
SuccessExitStatus=1
TimeoutStopSec=5
KillSignal=SIGTERM
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/efis-ems.service.d/performance.conf
[Unit]
Wants=efis-avionics-link-tuning.service
After=efis-avionics-link-tuning.service

[Service]
# CPU0 receives both bcmgenet Ethernet IRQs on the verified RPi4. Keeping all
# renderer threads on CPU1-3 removed second-scale latency bursts under eglfs.
CPUAffinity=1 2 3
